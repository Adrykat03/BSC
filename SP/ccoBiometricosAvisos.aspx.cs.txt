        private DataTable GetDataSource(string consulta)
        {
            string query = consulta;
            string conexion1 = DBLibreria.Generales.extraerValordeconfiguracion("CNUtilitario");
            DataSet ds = new DataSet();
            DataTable dt = new DataTable();

            using (SqlConnection conexion = new SqlConnection(conexion1))
            using (SqlDataAdapter adapter = new SqlDataAdapter(query.Trim(), conexion))
            {
                adapter.SelectCommand.CommandTimeout = 640;  
                conexion.Open();
                adapter.Fill(ds, "Resultado");
                conexion.Close();
            }

            Tabla1 = ds.Tables["Resultado"];
            return Tabla1;
        }
